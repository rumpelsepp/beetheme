Params (dict): terms the taxonomy, e.g. .Data.Terms of its page or site.Taxonomies.tags (required) skip titles of terms to leave out (optional) label what the numbers count, e.g. "Rezepte" (default: the text "tagCloudEntries" of i18n/) */ -}} {{- $skip := .skip | default slice }} {{- $label := .label | default (T "tagCloudEntries") }} {{- $terms := slice }} {{- range .terms.Alphabetical }} {{- if not (in $skip .Page.Title) }}{{ $terms = $terms | append . }}{{ end }} {{- end }} {{- with $terms }} {{- $min := (index . 0).Count }} {{- $max := $min }} {{- range . }} {{- if lt .Count $min }}{{ $min = .Count }}{{ end }} {{- if gt .Count $max }}{{ $max = .Count }}{{ end }} {{- end }} {{- $lmin := math.Log (math.Max $min 1) }} {{- $lmax := math.Log (math.Max $max 1) }} {{- $span := cond (gt $lmax $lmin) (sub $lmax $lmin) 1.0 }}
